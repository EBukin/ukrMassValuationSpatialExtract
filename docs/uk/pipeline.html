Конвеєр обробки

Проєкт складається з коротких пронумерованих скриптів R у корені проєкту. Кожен запускається через source() з кореня проєкту і спершу підключає 00-setup.R. Кожен скрипт вилучення записує одну довгу таблицю до data/interim/; 21-combine.R об’єднує їх у підсумкову таблицю.

Час орієнтовний, за одним запуском на ноутбуці з Windows, з GLO-30 і растрами на локальному диску. Повний запуск без завантажень триває близько 3–4 хвилин.

Потік даних

Правила побудови коду

  • Простий код для початківців. tidyverse, sf, terra та exactextractr; короткі скрипти й мало коментарів. Жодних допоміжних функцій, коду пакетів чи регулярних виразів. Шляхи до файлів, позначки років та імена змінних записано повністю, а кожен рік посівного покриву має власний блок.
  • Жодного перепроєктування чи передискретизації растрів. Натомість кожне вилучення трансформує ділянки до власної системи координат растра. Єдині растрові операції — віртуальна мозаїка ЦМВ, crop(), попіксельна арифметика над растрами зі спільною сіткою та terrain().
  • Проєкції. crs_work <- "EPSG:3035" (рівновелика проєкція, метри) використовується для побудови форм ділянок, розрахунку area_ha та побудови карт. Відстані вимірюються на сфері, в EPSG:4326.
  • Одна довга таблиця. Стовпці parcel_id | variable | value | source. Кожен метод вимірювання є окремою змінною (_mean, _centroid, _edge, _max, _range).
  • Контрольні перевірки. Скрипти зупиняються через cli_abort(), коли перевірка не проходить: кількість рядків, пропущені значення, діапазони значень.

Скрипт завантаження

# 00b-download.R
# In:  internet only
#      - OCHA COD-AB Ukraine admin boundaries, shapefile zip (81.5 MB), licence CC BY-IGO
#        https://data.humdata.org/dataset/cod-ab-ukr
#        -> data/raw/cod-ab/ukr_admin_boundaries.shp.zip (ADM1 files unzipped next to it)
#      - Copernicus DEM GLO-30 (or GLO-90), tiles over Poltava oblast plus 10 km (about 12 tiles, 20-40 MB each for GLO-30)
#        licence: Copernicus DEM licence (free use, attribution "produced using Copernicus WorldDEM-30 (c) DLR e.V. 2010-2014 and (c) Airbus Defence and Space GmbH 2014-2018")
#        https://copernicus-dem-30m.s3.amazonaws.com (GLO-30), https://copernicus-dem-90m.s3.amazonaws.com (GLO-90)
#        -> data/raw/dem/GLO-30/<tile id>.tif
# Out: data/interim/oblasts.gpkg (all oblasts), data/interim/dem.vrt (virtual mosaic of the DEM tiles)
# Purpose: download the public inputs once; existing files are skipped.

source("00-setup.R")

dir.create(codab_dir, recursive = TRUE, showWarnings = FALSE)
dir.create(file.path(dem_dir, dem_product), recursive = TRUE, showWarnings = FALSE)

h <- curl::new_handle(ssl_options = 2)  # this network cannot check certificate revocation

# COD-AB oblast boundaries (ADM1) ---------------------------------------------

codab_url <- "https://data.humdata.org/dataset/d23f529f-31e4-4021-a65b-13987e5cfb42/resource/2b38131d-5dd6-4b0f-b4c6-4aecc67f179d/download/ukr_admin_boundaries.shp.zip"
codab_zip <- file.path(codab_dir, "ukr_admin_boundaries.shp.zip")
codab_shp <- file.path(codab_dir, "ukr_admin1.shp")

if (!file.exists(codab_zip)) curl::curl_download(codab_url, codab_zip, handle = h)
if (!file.exists(codab_shp)) {
  unzip(
    codab_zip,
    files = c("ukr_admin1.shp", "ukr_admin1.shx", "ukr_admin1.dbf", "ukr_admin1.prj", "ukr_admin1.cpg"),
    exdir = codab_dir
  )
}

oblasts_raw <- st_read(codab_shp, quiet = TRUE)
cli_inform("COD-AB ADM1 fields: {.field {names(oblasts_raw)}}")

oblasts <- oblasts_raw |>
  select(oblast_pcode = adm1_pcode, oblast_ua = adm1_name1, oblast_en = adm1_name) |>
  st_transform(4326)

cli_inform("Renamed {.field adm1_pcode}, {.field adm1_name1}, {.field adm1_name} to {.field oblast_pcode}, {.field oblast_ua}, {.field oblast_en}.")
cli_inform("{nrow(oblasts)} oblasts read (27 expected).")

study_oblast <- oblasts |> filter(oblast_en == study_oblast_en)
if (nrow(study_oblast) != 1) cli_abort("Study oblast {study_oblast_en} not found in COD-AB.")

st_write(oblasts, oblasts_path, delete_dsn = TRUE, quiet = TRUE)
cli_inform("Wrote {.path {oblasts_path}}.")

# Copernicus DEM tiles --------------------------------------------------------

dem_products <- tribble(
  ~product, ~bucket,                                         ~code,
  "GLO-30", "https://copernicus-dem-30m.s3.amazonaws.com",  "10",
  "GLO-90", "https://copernicus-dem-90m.s3.amazonaws.com",  "30"
) |> filter(product == dem_product)

# 1 x 1 degree tile boxes over Ukraine, kept where they touch Poltava plus 10 km
study_buffer <- st_buffer(study_oblast, 10000)
tiles <- st_make_grid(cellsize = 1, offset = c(22, 44), n = c(19, 9), crs = 4326)
tiles <- tiles[lengths(st_intersects(tiles, study_buffer)) > 0]

# Lower-left corner of each tile, from its centre
tile_centres <- st_coordinates(st_centroid(tiles))
lon <- round(tile_centres[, "X"] - 0.5)
lat <- round(tile_centres[, "Y"] - 0.5)

dem_tiles <- tibble(
  id = str_c("Copernicus_DSM_COG_", dem_products$code, "_N", lat, "_00_E", sprintf("%03d", lon), "_00_DEM"),
  url = str_c(dem_products$bucket, "/", id, "/", id, ".tif"),
  dest = file.path(dem_dir, dem_product, str_c(id, ".tif"))
)
cli_inform("{nrow(dem_tiles)} {dem_product} tiles intersect {study_oblast_en} plus 10 km.")

# Download only the tiles not on disk yet (multi_download takes handle options, not a handle)
to_get <- dem_tiles |> filter(!file.exists(dest))
cli_inform("{nrow(to_get)} tiles to download.")
res <- curl::multi_download(to_get$url, to_get$dest, progress = FALSE, ssl_options = 2)

# A failed request (e.g. 404 for a sea tile) leaves an error page on disk: remove it
failed <- res |> filter(status_code != 200)
file.remove(failed$destfile[file.exists(failed$destfile)])
if (nrow(failed) > 0) cli_warn("{nrow(failed)} tile{?s} not downloaded: {.url {failed$url}}")

files_ok <- dem_tiles$dest[file.exists(dem_tiles$dest)]
cli_inform("{length(files_ok)} of {nrow(dem_tiles)} tiles on disk.")

# The mosaic uses the finest tile grid (1 arc-second). Tiles north of 50 N have coarser
# longitude spacing; GDAL repeats their pixels on that grid on the fly (nearest neighbour)
dem <- vrt(files_ok, filename = dem_vrt_path, options = c("-resolution", "highest"), overwrite = TRUE)
cli_inform("Wrote {.path {dem_vrt_path}}.")

Скрипт об’єднання

21-combine.R об’єднує п’ять довгих таблиць і зупиняється, якщо будь-яка пара ділянка × змінна трапляється двічі, якщо будь-яке значення пропущене, якщо source має значення поза чотирма рівнями або якщо ділянка не має рівно 81 змінної.

# 21-combine.R
# In:  data/interim/long_distance.parquet, long_soil.parquet, long_relief.parquet, long_crop.parquet, long_crop_2025.parquet
# Out: output/tables/spatial_covariates.parquet, output/tables/spatial_covariates.csv
# Purpose: bind the five long tables into one table of 81 covariates per parcel and check it.

source("00-setup.R")

long_distance <- arrow::read_parquet(long_distance_path)
long_soil <- arrow::read_parquet(long_soil_path)
long_relief <- arrow::read_parquet(long_relief_path)
long_crop <- arrow::read_parquet(long_crop_path)
long_crop_2025 <- arrow::read_parquet(long_crop_2025_path)

covariates <- bind_rows(long_distance, long_soil, long_relief, long_crop, long_crop_2025) |>
  select(parcel_id, variable, value, source) |>
  arrange(parcel_id, variable)

# Checks ----------------------------------------------------------------------

n_duplicates <- covariates |> count(parcel_id, variable) |> filter(n > 1) |> nrow()
if (n_duplicates > 0) cli_abort("{n_duplicates} parcel x variable combinations appear more than once.")

if (anyNA(covariates)) cli_abort("The combined table has missing values.")

if (!all(covariates$source %in% c("observed", "h3_6", "h3_5", "oblast"))) cli_abort("{.field source} has values outside observed, h3_6, h3_5, oblast.")

variables_per_parcel <- covariates |> count(parcel_id)
if (any(variables_per_parcel$n != 81)) cli_abort("Not every parcel has 81 variables.")
if (n_distinct(covariates$variable) != 81) cli_abort("The table does not have 81 distinct variables.")

# Summary ---------------------------------------------------------------------

cli_inform("Parcels: {n_distinct(covariates$parcel_id)}")
cli_inform("Variables: {n_distinct(covariates$variable)}")
cli_inform("Rows: {nrow(covariates)}")
print(count(covariates, source))

# Write -----------------------------------------------------------------------

gc()
arrow::write_parquet(covariates, covariates_parquet_path)
write_csv(covariates, covariates_csv_path)

cli_inform("Wrote {nrow(covariates)} rows to {.path {covariates_parquet_path}} and {.path {covariates_csv_path}}.")

Скрипт описової статистики

22-descriptives.R записує зведені таблиці, наведені на сторінках показників і на сторінці Результати.

# 22-descriptives.R
# In:  output/tables/spatial_covariates.parquet, data/clean/reference/parcels.parquet
# Out: output/tables/soil_centroid_vs_polygon.csv, crop_by_year.csv, distance_summary.csv, relief_summary.csv,
#      soil_source_by_case.csv; output/figures/soil_centroid_vs_polygon.png
# Purpose: descriptive statistics of the covariates, printed and saved.

source("00-setup.R")

covariates <- arrow::read_parquet(covariates_parquet_path)
parcels <- arrow::read_parquet(parcels_path, col_select = c(parcel_id, case))   # attributes only, no geometry

# Soil: centroid value vs polygon mean -----------------------------------------

# Units are SoilGrids mapped units (see 04-soil.R)
soil_pairs <- tribble(
  ~property,  ~mean_var,                       ~centroid_var,                       ~label,
  "bdod",     "soil_bdod_0_30cm_mean",         "soil_bdod_0_30cm_centroid",         "bdod (cg/cm³)",
  "cec",      "soil_cec_0_30cm_mean",          "soil_cec_0_30cm_centroid",          "cec (mmol(c)/kg)",
  "cfvo",     "soil_cfvo_0_30cm_mean",         "soil_cfvo_0_30cm_centroid",         "cfvo (cm³/dm³)",
  "clay",     "soil_clay_0_30cm_mean",         "soil_clay_0_30cm_centroid",         "clay (g/kg)",
  "nitrogen", "soil_nitrogen_0_30cm_mean",     "soil_nitrogen_0_30cm_centroid",     "nitrogen (cg/kg)",
  "ocd",      "soil_ocd_0_30cm_mean",          "soil_ocd_0_30cm_centroid",          "ocd (hg/m³)",
  "ocs",      "soil_ocs_0_30cm_mean",          "soil_ocs_0_30cm_centroid",          "ocs (t/ha)",
  "phh2o",    "soil_phh2o_0_30cm_mean",        "soil_phh2o_0_30cm_centroid",        "phh2o (pH × 10)",
  "sand",     "soil_sand_0_30cm_mean",         "soil_sand_0_30cm_centroid",         "sand (g/kg)",
  "silt",     "soil_silt_0_30cm_mean",         "soil_silt_0_30cm_centroid",         "silt (g/kg)"
)

soil_mean <- covariates |>
  inner_join(soil_pairs, by = c("variable" = "mean_var")) |>
  select(parcel_id, property, label, mean_value = value, mean_source = source)

soil_centroid <- covariates |>
  inner_join(soil_pairs, by = c("variable" = "centroid_var")) |>
  select(parcel_id, property, centroid_value = value, centroid_source = source)

soil_compare <- soil_mean |>
  inner_join(soil_centroid, by = c("parcel_id", "property")) |>
  mutate(observed = mean_source == "observed" & centroid_source == "observed")

if (nrow(soil_compare) != 10 * n_distinct(covariates$parcel_id)) cli_abort("Soil pairs do not cover 10 properties for every parcel.")

soil_stats_observed <- soil_compare |>
  filter(observed) |>
  group_by(property, label) |>
  summarise(
    rows = "observed",
    pearson = cor(mean_value, centroid_value),
    spearman = cor(mean_value, centroid_value, method = "spearman"),
    mean_abs_diff = mean(abs(centroid_value - mean_value)),
    n = n(),
    .groups = "drop"
  )

soil_stats_all <- soil_compare |>
  group_by(property, label) |>
  summarise(
    rows = "all",
    pearson = cor(mean_value, centroid_value),

    mean_abs_diff = mean(abs(centroid_value - mean_value)),
    n = n(),
    .groups = "drop"
  )

soil_stats <- bind_rows(soil_stats_observed, soil_stats_all) |>
  arrange(property, desc(rows))

if (anyNA(soil_stats$pearson) || anyNA(soil_stats$spearman)) cli_abort("Some soil correlations are missing.")
if (any(abs(soil_stats$pearson) > 1) || any(abs(soil_stats$spearman) > 1)) cli_abort("Soil correlations outside [-1, 1].")
